// Проверка статус кода
pm.test("Status code is 200", function () {
    pm.response.to.have.status(200);
});

// Проверка на то что возвращается объект с id=1
 pm.test("Returning object contains field id=1", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.id).to.eql("1");
});

// Проверка на то что родительский id равен 113
pm.test("Parent id is 113", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.parent_id).to.eql("113");
});
